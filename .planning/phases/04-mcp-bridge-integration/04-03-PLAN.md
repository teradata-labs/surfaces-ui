```typescript
/**
 * Action Handler Tests
 *
 * Unit tests for MCP action bridge layer.
 * Tests verify:
 * - MCP-02: execute_sql maps to bridge.callTool
 * - MCP-03: State coordination with functional updates
 * - MCP-04: Async result handling (loading/error states)
 */
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { createMcpActionHandlers, type McpActionState } from './actions';
import type { BridgeLike } from '../bridge';

// Mock bridge that implements BridgeLike interface
function createMockBridge(callToolResponse?: {
  success: boolean;
  data?: unknown;
  error?: string;
}): BridgeLike {
  return {
    connect: vi.fn(),
    disconnect: vi.fn(),
    isConnected: true,
    callTool: vi.fn().mockResolvedValue(
      callToolResponse ?? { success: true, data: { columns: [], rows: [] } }
    ),
    onDataPush: vi.fn(() => () => {}),
    fetchPage: vi.fn().mockResolvedValue({ rows: [], total: 0, offset: 0, limit: 10, hasMore: false }),
    onStreamChunk: vi.fn(() => () => {}),
    startStream: vi.fn(() => ({ id: 1, result: Promise.resolve({ success: true }) })),
    cancelStream: vi.fn(),
  };
}

describe('createMcpActionHandlers', () => {
  let mockBridge: BridgeLike;
  let handlers: ReturnType<typeof createMcpActionHandlers>;
  let setState: ReturnType<typeof vi.fn>;
  let stateUpdates: McpActionState[];

  beforeEach(() => {
    mockBridge = createMockBridge();
    handlers = createMcpActionHandlers(mockBridge);
    stateUpdates = [];
    // Track all state updates by capturing the updater function results
    setState = vi.fn((updater) => {
      const prevState = stateUpdates[stateUpdates.length - 1] ?? {};
      const newState = typeof updater === 'function' ? updater(prevState) : updater;
      stateUpdates.push(newState);
    });
  });

  describe('execute_sql', () => {
    it('calls bridge.callTool with correct params', async () => {
      const params = { query: 'SELECT * FROM users', connection_id: 'conn-123' };
      await handlers.execute_sql(params, setState, {});

      expect(mockBridge.callTool).toHaveBeenCalledWith('execute_sql', {
        query: 'SELECT * FROM users',
        connection_id: 'conn-123',
      });
    });

    it('sets loading state before calling bridge', async () => {
      await handlers.execute_sql({ query: 'SELECT 1' }, setState, {});

      // First state update should set loading: true
      expect(stateUpdates[0]).toMatchObject({
        loading: true,
        error: null,
      });
    });

    it('transforms MCP result format to json-render state format', async () => {
      const mcpResult = {
        success: true,
        data: {
          columns: ['id', 'name', 'email'],
          rows: [
            [1, 'Alice', 'alice@example.com'],
            [2, 'Bob', 'bob@example.com'],
          ],
        },
      };
      const bridge = createMockBridge(mcpResult);
      const h = createMcpActionHandlers(bridge);

      await h.execute_sql({ query: 'SELECT * FROM users' }, setState, {});

      // Final state should have transformed data
      const finalState = stateUpdates[stateUpdates.length - 1];
      expect(finalState.data).toEqual([
        { id: 1, name: 'Alice', email: 'alice@example.com' },
        { id: 2, name: 'Bob', email: 'bob@example.com' },
      ]);
      expect(finalState.columns).toEqual(['id', 'name', 'email']);
      expect(finalState.row_count).toBe(2);
      expect(finalState.loading).toBe(false);
      expect(finalState.error).toBeNull();
    });

    it('sets error state on bridge failure', async () => {
      const bridge = createMockBridge({ success: false, error: 'Connection refused' });
      const h = createMcpActionHandlers(bridge);

      await h.execute_sql({ query: 'SELECT 1' }, setState, {});

      const finalState = stateUpdates[stateUpdates.length - 1];
      expect(finalState.error).toBe('Connection refused');
      expect(finalState.loading).toBe(false);
    });

    it('sets error state on thrown exception', async () => {
      const bridge = createMockBridge();
      (bridge.callTool as ReturnType<typeof vi.fn>).mockRejectedValue(new Error('Network error'));
      const h = createMcpActionHandlers(bridge);

      await h.execute_sql({ query: 'SELECT 1' }, setState, {});

      const finalState = stateUpdates[stateUpdates.length - 1];
      expect(finalState.error).toBe('Network error');
      expect(finalState.loading).toBe(false);
    });

    it('uses functional setState for atomic updates (MCP-03)', async () => {
      await handlers.execute_sql({ query: 'SELECT 1' }, setState, {});

      // Verify setState was called with functions, not values
      const calls = setState.mock.calls;
      expect(typeof calls[0][0]).toBe('function'); // loading: true
      expect(typeof calls[1][0]).toBe('function'); // result/error
    });

    it('preserves existing state during updates', async () => {
      const existingState: McpActionState = {
        data: [{ old: 'data' }],
        columns: ['old'],
        row_count: 1,
        loading: false,
        error: null,
      };
      stateUpdates.push(existingState);

      await handlers.execute_sql({ query: 'SELECT 1' }, setState, existingState);

      // Loading state should preserve other fields
      expect(stateUpdates[1]).toMatchObject({
        data: [{ old: 'data' }], // preserved
        columns: ['old'],        // preserved
        loading: true,
        error: null,
      });
    });
  });

  describe('export_csv', () => {
    it('generates CSV from state data', async () => {
      // Mock document.createElement and URL.createObjectURL
      const mockLink = { href: '', download: '', click: vi.fn() };
      vi.spyOn(document, 'createElement').mockReturnValue(mockLink as unknown as HTMLAnchorElement);
      vi.spyOn(URL, 'createObjectURL').mockReturnValue('blob:test');
      vi.spyOn(URL, 'revokeObjectURL').mockImplementation(() => {});

      const state: McpActionState = {
        data: [
          { id: 1, name: 'Alice' },
          { id: 2, name: 'Bob' },
        ],
        columns: ['id', 'name'],
      };

      await handlers.export_csv({ filename: 'test.csv' }, setState, state);

      expect(mockLink.download).toBe('test.csv');
      expect(mockLink.click).toHaveBeenCalled();
    });
  });

  describe('copy_json', () => {
    it('copies state data to clipboard as JSON', async () => {
      const writeTextMock = vi.fn().mockResolvedValue(undefined);
      Object.assign(navigator, { clipboard: { writeText: writeTextMock } });

      const state: McpActionState = {
        data: [{ id: 1, name: 'Test' }],
      };

      await handlers.copy_json({}, setState, state);

      expect(writeTextMock).toHaveBeenCalledWith(
        JSON.stringify([{ id: 1, name: 'Test' }], null, 2)
      );
    });
  });
});
```

- Error handling for bridge failures and exceptions

```typescript
/**
 * Renderer Integration Tests
 *
 * Integration tests for SurfacesRenderer with MCP Bridge.
 * Tests verify:
 * - MCP-05: Bridge context passed to registry components
 * - Action handlers wired correctly
 * - Backward compatibility without bridge
 */
import React from 'react';
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { render, screen, waitFor } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { SurfacesRenderer } from './renderer';
import type { BridgeLike } from '../bridge';
import type { Spec } from '@json-render/react';

// Mock bridge implementation
function createMockBridge(): BridgeLike {
  return {
    connect: vi.fn(),
    disconnect: vi.fn(),
    isConnected: true,
    callTool: vi.fn().mockResolvedValue({
      success: true,
      data: {
        columns: ['id', 'name'],
        rows: [[1, 'Test']],
      },
    }),
    onDataPush: vi.fn(() => () => {}),
    fetchPage: vi.fn().mockResolvedValue({ rows: [], total: 0, offset: 0, limit: 10, hasMore: false }),
    onStreamChunk: vi.fn(() => () => {}),
    startStream: vi.fn(() => ({ id: 1, result: Promise.resolve({ success: true }) })),
    cancelStream: vi.fn(),
  };
}

describe('SurfacesRenderer with MCP Bridge', () => {
  let mockBridge: BridgeLike;

  beforeEach(() => {
    mockBridge = createMockBridge();
  });

  it('renders without bridge (backward compatible)', () => {
    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: { placeholder: 'Search...' },
        },
      },
    };

    render(<SurfacesRenderer spec={spec} />);

    expect(screen.getByPlaceholderText('Search...')).toBeTruthy();
  });

  it('renders with bridge provided', () => {
    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: { placeholder: 'With Bridge' },
        },
      },
    };

    render(<SurfacesRenderer spec={spec} bridge={mockBridge} />);

    expect(screen.getByPlaceholderText('With Bridge')).toBeTruthy();
  });

  it('merges user handlers with MCP handlers', () => {
    const customHandler = vi.fn();
    const handlers = { custom_action: customHandler };

    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: {},
        },
      },
    };

    // This test verifies the merge happens without errors
    // Actual handler invocation tested in action tests
    expect(() => {
      render(
        <SurfacesRenderer
          spec={spec}
          bridge={mockBridge}
          handlers={handlers}
        />
      );
    }).not.toThrow();
  });

  it('creates MCP action handlers when bridge provided', async () => {
    // Test that the action handlers are created and functional
    // by rendering a DataTable spec that could trigger actions
    const spec: Spec = {
      root: 'table',
      elements: {
        table: {
          type: 'DataTable',
          props: {
            columns: ['id', 'name'],
            height: 300,
          },
        },
      },
    };

    render(
      <SurfacesRenderer
        spec={spec}
        bridge={mockBridge}
        initialState={{ data: [{ id: 1, name: 'Test' }] }}
      />
    );

    // Component renders with initial data
    await waitFor(() => {
      expect(screen.getByText('id')).toBeTruthy();
      expect(screen.getByText('name')).toBeTruthy();
    });
  });

  it('preserves theme prop with bridge', () => {
    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: {},
        },
      },
    };

    // Theme prop should still work when bridge is provided
    expect(() => {
      render(
        <SurfacesRenderer
          spec={spec}
          bridge={mockBridge}
          theme="dark"
        />
      );
    }).not.toThrow();
  });

  it('preserves onStateChange callback with bridge', () => {
    const onStateChange = vi.fn();
    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: { placeholder: 'Type here' },
        },
      },
    };

    render(
      <SurfacesRenderer
        spec={spec}
        bridge={mockBridge}
        onStateChange={onStateChange}
      />
    );

    // Renderer should accept onStateChange without errors
    expect(screen.getByPlaceholderText('Type here')).toBeTruthy();
  });
});

describe('SurfacesRenderer without MCP Bridge', () => {
  it('uses only user-provided handlers when no bridge', () => {
    const customHandler = vi.fn();
    const handlers = { execute_sql: customHandler };

    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: {},
        },
      },
    };

    // Without bridge, custom execute_sql handler should be used
    // (not the MCP one that would call bridge.callTool)
    expect(() => {
      render(
        <SurfacesRenderer
          spec={spec}
          handlers={handlers}
        />
      );
    }).not.toThrow();
  });

  it('renders with empty handlers', () => {
    const spec: Spec = {
      root: 'main',
      elements: {
        main: {
          type: 'Filter',
          props: { placeholder: 'No handlers' },
        },
      },
    };

    render(<SurfacesRenderer spec={spec} handlers={{}} />);

    expect(screen.getByPlaceholderText('No handlers')).toBeTruthy();
  });
});
```
