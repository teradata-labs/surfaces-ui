   ```python
   """LLM provider abstraction for UI generation."""
   import os
   from abc import ABC, abstractmethod
   from dataclasses import dataclass
   from typing import AsyncIterator, Optional

   @dataclass
   class LLMResponse:
       """Response from LLM provider."""
       content: str
       model: str
       provider: str
       usage: Optional[dict] = None

   class LLMProvider(ABC):
       """Abstract base class for LLM providers."""

       @abstractmethod
       async def generate(
           self,
           system_prompt: str,
           user_prompt: str,
           json_schema: Optional[dict] = None,
       ) -> LLMResponse:
           """Generate a response from the LLM."""
           pass

       @abstractmethod
       async def stream(
           self,
           system_prompt: str,
           user_prompt: str,
       ) -> AsyncIterator[str]:
           """Stream response chunks from the LLM."""
           pass

       @abstractmethod
       def is_configured(self) -> bool:
           """Check if provider has required credentials."""
           pass

   def get_llm_provider() -> LLMProvider:
       """Factory function to get configured LLM provider.

       Uses LLM_PROVIDER env var: 'bedrock' | 'anthropic' | 'openai'
       Defaults to 'bedrock' if not set.
       """
       provider_name = os.environ.get("LLM_PROVIDER", "bedrock").lower()

       if provider_name == "bedrock":
           from .bedrock import BedrockProvider
           return BedrockProvider()
       elif provider_name == "anthropic":
           from .anthropic_provider import AnthropicProvider
           return AnthropicProvider()
       elif provider_name == "openai":
           from .openai_provider import OpenAIProvider
           return OpenAIProvider()
       else:
           raise ValueError(f"Unknown LLM provider: {provider_name}. Use 'bedrock', 'anthropic', or 'openai'")
   ```

   ```python
   """AWS Bedrock provider for Claude models."""
   import json
   import logging
   import os
   from typing import AsyncIterator, Optional

   import boto3
   from botocore.config import Config

   from .provider import LLMProvider, LLMResponse

   logger = logging.getLogger(__name__)

   class BedrockProvider(LLMProvider):
       """AWS Bedrock provider using Claude models via Converse API."""

       MODEL_ID = "anthropic.claude-sonnet-4-5-20250514-v1:0"

       def __init__(self):
           self._client = None
           self._region = os.environ.get("AWS_REGION", "us-east-1")

       @property
       def client(self):
           if self._client is None:
               config = Config(
                   region_name=self._region,
                   retries={"max_attempts": 3, "mode": "adaptive"}
               )
               self._client = boto3.client("bedrock-runtime", config=config)
           return self._client

       def is_configured(self) -> bool:
           """Check if AWS credentials are available (SSO or env vars)."""
           try:
               # This will use SSO credentials if configured
               sts = boto3.client("sts")
               sts.get_caller_identity()
               return True
           except Exception:
               return False

       async def generate(
           self,
           system_prompt: str,
           user_prompt: str,
           json_schema: Optional[dict] = None,
       ) -> LLMResponse:
           """Generate using Bedrock Converse API with optional structured output."""
           messages = [{"role": "user", "content": [{"text": user_prompt}]}]

           inference_config = {
               "maxTokens": 4096,
               "temperature": 0,
           }

           # Build request
           request = {
               "modelId": self.MODEL_ID,
               "messages": messages,
               "system": [{"text": system_prompt}],
               "inferenceConfig": inference_config,
           }

           # Add tool for structured output if schema provided
           if json_schema:
               request["toolConfig"] = {
                   "tools": [{
                       "toolSpec": {
                           "name": "json_output",
                           "description": "Output the result as structured JSON",
                           "inputSchema": {"json": json_schema}
                       }
                   }],
                   "toolChoice": {"tool": {"name": "json_output"}}
               }

           try:
               response = self.client.converse(**request)

               # Extract content - check for tool use or text
               output = response.get("output", {})
               message = output.get("message", {})
               content_blocks = message.get("content", [])

               result_text = ""
               for block in content_blocks:
                   if "text" in block:
                       result_text = block["text"]
                   elif "toolUse" in block:
                       # Structured output via tool
                       result_text = json.dumps(block["toolUse"].get("input", {}))

               return LLMResponse(
                   content=result_text,
                   model=self.MODEL_ID,
                   provider="bedrock",
                   usage=response.get("usage"),
               )
           except Exception as e:
               logger.error(f"Bedrock generation failed: {e}")
               raise

       async def stream(
           self,
           system_prompt: str,
           user_prompt: str,
       ) -> AsyncIterator[str]:
           """Stream using Bedrock Converse Stream API."""
           messages = [{"role": "user", "content": [{"text": user_prompt}]}]

           try:
               response = self.client.converse_stream(
                   modelId=self.MODEL_ID,
                   messages=messages,
                   system=[{"text": system_prompt}],
                   inferenceConfig={"maxTokens": 4096, "temperature": 0},
               )

               for event in response.get("stream", []):
                   if "contentBlockDelta" in event:
                       delta = event["contentBlockDelta"].get("delta", {})
                       if "text" in delta:
                           yield delta["text"]
           except Exception as e:
               logger.error(f"Bedrock streaming failed: {e}")
               raise
   ```

   ```python
   """Anthropic direct API provider."""
   import logging
   import os
   from typing import AsyncIterator, Optional

   from anthropic import Anthropic

   from .provider import LLMProvider, LLMResponse

   logger = logging.getLogger(__name__)

   class AnthropicProvider(LLMProvider):
       """Anthropic direct API provider."""

       MODEL = "claude-sonnet-4-5-20250514"

       def __init__(self):
           self._client = None

       @property
       def client(self):
           if self._client is None:
               api_key = os.environ.get("ANTHROPIC_API_KEY")
               if not api_key:
                   raise ValueError("ANTHROPIC_API_KEY environment variable not set")
               self._client = Anthropic(api_key=api_key)
           return self._client

       def is_configured(self) -> bool:
           return bool(os.environ.get("ANTHROPIC_API_KEY"))

       async def generate(
           self,
           system_prompt: str,
           user_prompt: str,
           json_schema: Optional[dict] = None,
       ) -> LLMResponse:
           """Generate using Anthropic API with optional structured output."""
           messages = [{"role": "user", "content": user_prompt}]

           # Build request
           request = {
               "model": self.MODEL,
               "max_tokens": 4096,
               "system": system_prompt,
               "messages": messages,
           }

           # Add tool for structured output if schema provided
           if json_schema:
               request["tools"] = [{
                   "name": "json_output",
                   "description": "Output the result as structured JSON",
                   "input_schema": json_schema
               }]
               request["tool_choice"] = {"type": "tool", "name": "json_output"}

           try:
               response = self.client.messages.create(**request)

               # Extract content
               result_text = ""
               for block in response.content:
                   if block.type == "text":
                       result_text = block.text
                   elif block.type == "tool_use":
                       import json
                       result_text = json.dumps(block.input)

               return LLMResponse(
                   content=result_text,
                   model=self.MODEL,
                   provider="anthropic",
                   usage={"input_tokens": response.usage.input_tokens, "output_tokens": response.usage.output_tokens},
               )
           except Exception as e:
               logger.error(f"Anthropic generation failed: {e}")
               raise

       async def stream(
           self,
           system_prompt: str,
           user_prompt: str,
       ) -> AsyncIterator[str]:
           """Stream using Anthropic streaming API."""
           try:
               with self.client.messages.stream(
                   model=self.MODEL,
                   max_tokens=4096,
                   system=system_prompt,
                   messages=[{"role": "user", "content": user_prompt}],
               ) as stream:
                   for text in stream.text_stream:
                       yield text
           except Exception as e:
               logger.error(f"Anthropic streaming failed: {e}")
               raise
   ```

   ```python
   """OpenAI API provider."""
   import json
   import logging
   import os
   from typing import AsyncIterator, Optional

   from openai import OpenAI

   from .provider import LLMProvider, LLMResponse

   logger = logging.getLogger(__name__)

   class OpenAIProvider(LLMProvider):
       """OpenAI API provider using GPT-4."""

       MODEL = "gpt-4o"

       def __init__(self):
           self._client = None

       @property
       def client(self):
           if self._client is None:
               api_key = os.environ.get("OPENAI_API_KEY")
               if not api_key:
                   raise ValueError("OPENAI_API_KEY environment variable not set")
               self._client = OpenAI(api_key=api_key)
           return self._client

       def is_configured(self) -> bool:
           return bool(os.environ.get("OPENAI_API_KEY"))

       async def generate(
           self,
           system_prompt: str,
           user_prompt: str,
           json_schema: Optional[dict] = None,
       ) -> LLMResponse:
           """Generate using OpenAI API with optional structured output."""
           messages = [
               {"role": "system", "content": system_prompt},
               {"role": "user", "content": user_prompt},
           ]

           # Build request
           request = {
               "model": self.MODEL,
               "messages": messages,
               "max_tokens": 4096,
               "temperature": 0,
           }

           # Use response_format for structured output if schema provided
           if json_schema:
               request["response_format"] = {
                   "type": "json_schema",
                   "json_schema": {
                       "name": "ui_spec",
                       "schema": json_schema,
                       "strict": True
                   }
               }

           try:
               response = self.client.chat.completions.create(**request)

               result_text = response.choices[0].message.content or ""

               return LLMResponse(
                   content=result_text,
                   model=self.MODEL,
                   provider="openai",
                   usage={
                       "input_tokens": response.usage.prompt_tokens,
                       "output_tokens": response.usage.completion_tokens
                   } if response.usage else None,
               )
           except Exception as e:
               logger.error(f"OpenAI generation failed: {e}")
               raise

       async def stream(
           self,
           system_prompt: str,
           user_prompt: str,
       ) -> AsyncIterator[str]:
           """Stream using OpenAI streaming API."""
           try:
               stream = self.client.chat.completions.create(
                   model=self.MODEL,
                   messages=[
                       {"role": "system", "content": system_prompt},
                       {"role": "user", "content": user_prompt},
                   ],
                   max_tokens=4096,
                   temperature=0,
                   stream=True,
               )

               for chunk in stream:
                   if chunk.choices[0].delta.content:
                       yield chunk.choices[0].delta.content
           except Exception as e:
               logger.error(f"OpenAI streaming failed: {e}")
               raise
   ```

   ```python
   """Generate UI tool for MCP server."""
   import json
   import logging
   from typing import AsyncIterator, Optional

   from mcp.types import Tool

   from ..llm import get_llm_provider, LLMProvider

   logger = logging.getLogger(__name__)

   def get_generate_ui_tool_definition() -> Tool:
       """Return the MCP tool definition for generate_ui."""
       return Tool(
           name="generate_ui",
           description="Generate UI components from natural language descriptions. Returns a json-render specification that can be rendered directly.",
           inputSchema={
               "type": "object",
               "properties": {
                   "prompt": {
                       "type": "string",
                       "description": "Natural language description of desired UI"
                   },
                   "include_explanation": {
                       "type": "boolean",
                       "description": "Include reasoning about component choices",
                       "default": False
                   }
               },
               "required": ["prompt"]
           }
       )

   def get_catalog_system_prompt() -> str:
       """Return the system prompt for UI generation."""
       return '''You are a UI generation assistant. Generate json-render specifications for SurfacesUI components.

   ## Available Components

   ### Primitives
   - **Chart**: Data visualization (bar, line, pie, scatter, area, heatmap)
     Props: type, xAxis, yAxis, height, showDataLabels, colors
   - **DataTable**: Tabular data display with virtual scrolling
     Props: columns, height, density (compact|comfortable|spacious), virtualScrolling, showRowNumbers
   - **Filter**: Search input for filtering data
     Props: placeholder, debounceMs
   - **DensityToggle**: Row density selector
     Props: showLabels
   - **ViewModeToggle**: Table/chart view switcher
     Props: initialMode (table|chart)

   ### Molecules
   - **TableWithFilters**: DataTable + Filter + DensityToggle combined
     Props: columns, height, initialDensity, showFilter, showDensityToggle, virtualScrolling
   - **ChartWithControls**: Chart with type/axis configuration panel
     Props: height, initialChartType, showControls, controlsCollapsed

   ### Organisms
   - **SQLResultsDisplay**: Complete SQL results viewer with table, chart, export
     Props: sql, context (web|mcp), initialViewMode, tableHeight, chartHeight

   ## Output Format

   Return ONLY valid JSON matching this structure:
   {
     "root": "<element-id>",
     "elements": {
       "<element-id>": {
         "type": "<ComponentName>",
         "props": { ... }
       }
     }
   }

   ## Rules
   1. Use TableWithFilters for interactive data tables (not separate DataTable + Filter)
   2. Use SQLResultsDisplay for complete data exploration interfaces
   3. Use Chart for simple visualizations, ChartWithControls when users need options
   4. Default height is 400px unless specified
   5. Enable virtualScrolling for tables with potentially large datasets
   '''

   def get_ui_spec_json_schema() -> dict:
       """Return JSON schema for structured output."""
       return {
           "type": "object",
           "properties": {
               "root": {
                   "type": "string",
                   "description": "ID of the root element"
               },
               "elements": {
                   "type": "object",
                   "additionalProperties": {
                       "type": "object",
                       "properties": {
                           "type": {
                               "type": "string",
                               "enum": ["Chart", "DataTable", "Filter", "DensityToggle", "ViewModeToggle", "TableWithFilters", "ChartWithControls", "SQLResultsDisplay"]
                           },
                           "props": {
                               "type": "object"
                           }
                       },
                       "required": ["type", "props"]
                   }
               }
           },
           "required": ["root", "elements"]
       }

   async def generate_ui_tool(
       prompt: str,
       include_explanation: bool = False,
       streaming: bool = False,
   ) -> str:
       """Generate UI spec from natural language prompt.

       Args:
           prompt: Natural language description of desired UI
           include_explanation: Include reasoning about component choices
           streaming: Not used in MCP call_tool (streaming handled separately)

       Returns:
           JSON string with: success, spec, explanation (if requested), provider
       """
       try:
           provider = get_llm_provider()

           if not provider.is_configured():
               return json.dumps({
                   "success": False,
                   "error": f"LLM provider not configured. Set required environment variables for {type(provider).__name__}."
               })

           system_prompt = get_catalog_system_prompt()

           if include_explanation:
               # First call for reasoning
               explain_prompt = f"Explain your component choices for this UI request, then provide the json-render spec:\n\n{prompt}"
               response = await provider.generate(system_prompt, explain_prompt)

               # Parse explanation and spec from response
               # The response may contain both explanation and JSON
               content = response.content

               # Try to extract JSON from the response
               import re
               json_match = re.search(r'\{[\s\S]*"root"[\s\S]*"elements"[\s\S]*\}', content)
               if json_match:
                   spec_str = json_match.group()
                   explanation = content[:json_match.start()].strip()
                   spec = json.loads(spec_str)
               else:
                   # If no JSON found, make second call with structured output
                   spec_response = await provider.generate(
                       system_prompt,
                       prompt,
                       json_schema=get_ui_spec_json_schema()
                   )
                   spec = json.loads(spec_response.content)
                   explanation = content

               return json.dumps({
                   "success": True,
                   "spec": spec,
                   "explanation": explanation,
                   "provider": response.provider,
                   "model": response.model,
               })
           else:
               # Single call with structured output
               response = await provider.generate(
                   system_prompt,
                   prompt,
                   json_schema=get_ui_spec_json_schema()
               )

               spec = json.loads(response.content)

               return json.dumps({
                   "success": True,
                   "spec": spec,
                   "provider": response.provider,
                   "model": response.model,
               })

       except json.JSONDecodeError as e:
           logger.error(f"Failed to parse LLM response as JSON: {e}")
           return json.dumps({
               "success": False,
               "error": f"Invalid JSON in LLM response: {str(e)}"
           })
       except Exception as e:
           logger.error(f"generate_ui_tool failed: {e}")
           return json.dumps({
               "success": False,
               "error": str(e)
           })

   async def stream_ui_generation(
       prompt: str,
       include_explanation: bool = False,
   ) -> AsyncIterator[str]:
       """Stream UI generation for HTTP endpoint.

       Yields JSONL chunks for progressive updates.
       """
       try:
           provider = get_llm_provider()

           if not provider.is_configured():
               yield json.dumps({"error": "LLM provider not configured"}) + "\n"
               return

           system_prompt = get_catalog_system_prompt()

           # Stream the response
           full_content = ""
           async for chunk in provider.stream(system_prompt, prompt):
               full_content += chunk
               yield json.dumps({"chunk": chunk}) + "\n"

           # Parse final result
           import re
           json_match = re.search(r'\{[\s\S]*"root"[\s\S]*"elements"[\s\S]*\}', full_content)
           if json_match:
               spec = json.loads(json_match.group())
               yield json.dumps({
                   "done": True,
                   "success": True,
                   "spec": spec,
               }) + "\n"
           else:
               yield json.dumps({
                   "done": True,
                   "success": False,
                   "error": "Could not parse UI spec from response"
               }) + "\n"

       except Exception as e:
           logger.error(f"stream_ui_generation failed: {e}")
           yield json.dumps({"error": str(e)}) + "\n"
   ```

   ```python
   async def generate_ui_stream(request):
       """Stream UI generation via HTTP for progressive updates."""
       prompt = request.query.get('prompt', '')
       include_explanation = request.query.get('explain', 'false').lower() == 'true'

       if not prompt:
           return web.Response(
               text='{"error": "prompt required"}',
               status=400,
               content_type='application/json'
           )

       response = web.StreamResponse(
           status=200,
           reason='OK',
           headers={'Content-Type': 'application/x-ndjson'}
       )
       await response.prepare(request)

       async for chunk in stream_ui_generation(prompt, include_explanation):
           await response.write(chunk.encode('utf-8'))

       await response.write_eof()
       return response
   ```
