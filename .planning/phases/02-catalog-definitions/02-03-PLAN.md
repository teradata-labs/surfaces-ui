function CatalogSQLResultsDisplay(ctx: ComponentContext<any, 'SQLResultsDisplay'>) {
  const {
    sql,
    context: execContext,
    projectId,
    teradataBaseUrl,
    initialViewMode,
    tableHeight,
    chartHeight,
  } = (ctx.props as any).value;

  // Data comes from json-render context, not AI-specified props
  const rawData = (ctx as any).data?.value;

  // SQLResultsDisplay expects data in specific format
  // Support both {rows, columns} and {data, columns} formats
  const data = React.useMemo(() => {
    if (!rawData) return { rows: [], columns: [] };
    if (Array.isArray(rawData)) {
      // If just an array, derive columns from first row
      const columns = rawData.length > 0 ? Object.keys(rawData[0]) : [];
      return { rows: rawData, columns };
    }
    // Already in correct format
    return rawData;
  }, [rawData]);

  // Build savedConfig from AI-specifiable schema props
  const savedConfig = React.useMemo(() => ({
    resultsViewMode: initialViewMode || 'table',
    tableHeight: tableHeight || 400,
    chartHeight: chartHeight || 400,
  }), [initialViewMode, tableHeight, chartHeight]);

  return (
    <SQLResultsDisplay
      data={data}
      sql={sql}
      savedConfig={savedConfig}
      context={execContext}
      projectId={projectId}
      teradataBaseUrl={teradataBaseUrl}
    />
  );
}
```

export function generateCatalogPrompt(): string {
  const primitives = ['Chart', 'DataTable', 'Filter', 'DensityToggle', 'ViewModeToggle'];
  const molecules = ['TableWithFilters', 'ChartWithControls'];
  const organisms = ['SQLResultsDisplay'];

  let prompt = `# Available UI Components\n\n`;
  prompt += `Generate specifications for SQL results visualization.\n\n`;

  // Organisms section first (top-level recommendation)
  prompt += `## Organisms (Recommended for complete views)\n`;
  prompt += `Use organisms for full-featured UI sections:\n\n`;
  for (const name of organisms) {
    const def = catalog.components[name as keyof typeof catalog.components];
    prompt += formatComponent(name, def);
  }

  // Molecules section
  prompt += `## Molecules (For custom compositions)\n`;
  prompt += `Use molecules for integrated functionality:\n\n`;
  for (const name of molecules) {
    const def = catalog.components[name as keyof typeof catalog.components];
    prompt += formatComponent(name, def);
  }

  // Primitives section
  prompt += `## Primitives (For fine-grained control)\n`;
  prompt += `Use primitives for specific UI elements:\n\n`;
  for (const name of primitives) {
    const def = catalog.components[name as keyof typeof catalog.components];
    prompt += formatComponent(name, def);
  }

  // Actions section
  prompt += `# Available Actions\n\n`;
  for (const [name, def] of Object.entries(catalog.actions)) {
    prompt += `- ${name}: ${def.description}\n`;
  }

  return prompt;
}
```
