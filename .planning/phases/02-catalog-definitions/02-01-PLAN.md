<objective>
Refine primitive Zod schemas to contain only AI-specifiable props and create AI prompt generation utilities.

Purpose: Catalog schemas serve a different purpose than TypeScript interfaces. Components have three categories of props:

1. **AI-specifiable props** (IN catalog): Configuration that AI can determine from natural language
   - Example: chart type, height, density, placeholder text

2. **Context-provided props** (NOT in catalog): Data that comes from json-render context
   - Example: `data` array - always provided by parent context, not AI-specified

3. **Runtime props** (NOT in catalog): Callbacks and styling managed by parent components
   - Example: `onChange`, `onRowClick`, `className`

The catalog defines what AI can configure. The registry wrappers handle context/runtime props.

Output: Updated schema.ts with AI-specifiable primitive schemas, new prompts.ts with AI description utilities.
</objective>

/**
 * AI Prompt Generation Utilities for json-render Catalog
 *
 * Generates coherent, AI-consumable descriptions of components
 * for use in LLM prompts that generate UI specifications.
 *
 * Note: Catalog schemas only include AI-specifiable props.
 * Runtime props (callbacks, className) and context props (data)
 * are handled by registry wrappers, not specified by AI.
 */
import { catalog } from './schema';
import type { z } from 'zod';

/**
 * Generate a full catalog prompt describing all available components.
 * Used as system context for AI models generating UI specs.
 */
export function generateCatalogPrompt(): string {
  const components = Object.entries(catalog.components);
  const actions = Object.entries(catalog.actions);

  let prompt = `# Available UI Components\n\n`;
  prompt += `You can generate specifications for these components:\n\n`;

  for (const [name, def] of components) {
    prompt += `## ${name}\n`;
    prompt += `${def.description}\n\n`;
    prompt += `Props:\n`;

    // Extract prop descriptions from Zod schema
    const shape = (def.props as z.ZodObject<any>).shape;
    for (const [propName, propSchema] of Object.entries(shape)) {
      const desc = (propSchema as z.ZodTypeAny).description || 'No description';
      prompt += `- ${propName}: ${desc}\n`;
    }
    prompt += `\n`;
  }

  prompt += `# Available Actions\n\n`;
  for (const [name, def] of actions) {
    prompt += `- ${name}: ${def.description}\n`;
  }

  return prompt;
}

/**
 * Generate a focused prompt for a single component.
 * Used when the target component type is already known.
 */
export function generateComponentPrompt(componentName: keyof typeof catalog.components): string {
  const def = catalog.components[componentName];
  if (!def) {
    throw new Error(`Unknown component: ${componentName}`);
  }

  let prompt = `# ${componentName}\n\n`;
  prompt += `${def.description}\n\n`;
  prompt += `## Props\n\n`;

  const shape = (def.props as z.ZodObject<any>).shape;
  for (const [propName, propSchema] of Object.entries(shape)) {
    const zodSchema = propSchema as z.ZodTypeAny;
    const desc = zodSchema.description || 'No description';
    const isOptional = zodSchema.isOptional?.() || zodSchema._def?.defaultValue !== undefined;
    prompt += `- **${propName}**${isOptional ? ' (optional)' : ''}: ${desc}\n`;
  }

  return prompt;
}

/**
 * Generate a component usage example for AI context.
 */
export function generateComponentExample(componentName: keyof typeof catalog.components): string {
  const examples: Record<string, string> = {
    Chart: `{

    "type": "bar",
    "xAxis": "category",
    "yAxis": "value",
    "height": 400
  }
}`,
    DataTable: `{
  "type": "DataTable",
  "props": {
    "columns": ["id", "name", "email"],
    "height": 400,
    "density": "comfortable",
    "virtualScrolling": true
  }
}`,
    Filter: `{
  "type": "Filter",

    "placeholder": "Search users...",

  }
}`,

  "type": "DensityToggle",

}`,

  "type": "ViewModeToggle",
  "props": {
    "modes": ["table", "chart"],
    "activeMode": "table"
  }
}`,
  };

  return examples[componentName] || '{}';
}
