export function generateCatalogPrompt(): string {
  const primitives = ['Chart', 'DataTable', 'Filter', 'DensityToggle', 'ViewModeToggle'];
  const molecules = ['TableWithFilters', 'ChartWithControls'];

  let prompt = `# Available UI Components\n\n`;

  // Primitives section
  prompt += `## Primitives\nBasic building blocks:\n\n`;
  for (const name of primitives) {
    const def = catalog.components[name as keyof typeof catalog.components];
    prompt += formatComponent(name, def);
  }

  // Molecules section
  prompt += `## Molecules\nComposed components with integrated functionality:\n\n`;
  for (const name of molecules) {
    const def = catalog.components[name as keyof typeof catalog.components];
    prompt += formatComponent(name, def);
  }

  // Actions section
  prompt += `# Available Actions\n\n`;
  for (const [name, def] of Object.entries(catalog.actions)) {
    prompt += `- ${name}: ${def.description}\n`;
  }

  return prompt;
}

function formatComponent(name: string, def: any): string {
  let out = `### ${name}\n`;
  out += `${def.description}\n\n`;
  out += `Props:\n`;
  const shape = (def.props as z.ZodObject<any>).shape;
  for (const [propName, propSchema] of Object.entries(shape)) {
    const desc = (propSchema as z.ZodTypeAny).description || 'No description';
    out += `- ${propName}: ${desc}\n`;
  }
  out += `\n`;
  return out;
}
