/**
 * Type utilities for catalog-registry type safety
 *
 * Provides typed access to catalog schema props in registry wrappers,
 * eliminating the need for `as any` casts.
 */
import type { z } from 'zod';
import type { catalog } from './schema';

/* ── Generic Data Types ───────────────────────────────────────────────── */

/**
 * Generic row type for data arrays - record with string keys and unknown values
 */
export type DataRow = Record<string, unknown>;

/* ── Catalog Type Inference ───────────────────────────────────────────── */

/**
 * Extract component names from catalog
 */
export type CatalogComponentName = keyof typeof catalog.components;

/**
 * Extract props type for a specific catalog component
 * Uses z.infer to get the TypeScript type from the Zod schema
 */
export type CatalogComponentProps<K extends CatalogComponentName> =
  z.infer<(typeof catalog.components)[K]['props']>;

/* ── Chart Props ─────────────────────────────────────────────────────── */

export interface ChartProps {
  type: 'bar' | 'line' | 'area' | 'scatter' | 'pie';
  xAxis: string;
  yAxis: string;
  theme?: 'default' | 'brand' | 'financial';
  height?: number;
  showDataLabels?: boolean;
}

/* ── DataTable Props ─────────────────────────────────────────────────── */

export interface DataTableProps {
  columns: string[];
  height?: number;
  density?: 'compact' | 'comfortable' | 'spacious';
  virtualScrolling?: boolean;
}

/* ── Filter Props ────────────────────────────────────────────────────── */

export interface FilterProps {
  placeholder?: string;
  columns?: string[];
}

/* ── DensityToggle Props ─────────────────────────────────────────────── */

export interface DensityToggleProps {
  value?: 'compact' | 'comfortable' | 'spacious';
}

/* ── ViewModeToggle Props ────────────────────────────────────────────── */

export interface ViewModeToggleProps {
  modes?: ('table' | 'chart')[];
  activeMode?: 'table' | 'chart';
}

/* ── TableWithFilters Props ──────────────────────────────────────────── */

export interface TableWithFiltersProps {
  columns: string[];
  height?: number;
  initialDensity?: 'compact' | 'comfortable' | 'spacious';
  showFilter?: boolean;
  showDensityToggle?: boolean;
  virtualScrolling?: boolean;
}

/* ── ChartWithControls Props ─────────────────────────────────────────── */

export interface ChartWithControlsProps {
  columns: string[];
  height?: number;
  initialChartType?: 'bar' | 'line' | 'area' | 'scatter' | 'pie';
  initialXAxis?: string;
  initialYAxis?: string;
  showControls?: boolean;
  controlsCollapsed?: boolean;
}

/* ── SQLResultsDisplay Props ─────────────────────────────────────────── */

export interface SQLResultsDisplayProps {
  sql?: string;
  context?: string;
  projectId?: string;
  teradataBaseUrl?: string;
  initialViewMode?: 'table' | 'chart';
  tableHeight?: number;
  chartHeight?: number;
}
```

The interface definitions mirror the Zod schemas in schema.ts but provide explicit TypeScript types that can be used in registry wrappers without needing `z.infer` at runtime.

/**
 * Type-level tests for catalog schema and registry alignment
 *
 * These tests use TypeScript's type system to verify at compile time that:
 * 1. Catalog schemas match component prop types
 * 2. Registry wrapper types align with catalog definitions
 *
 * If these tests fail to compile, the types are misaligned.
 */
import { describe, it, expectTypeOf } from 'vitest';
import type { z } from 'zod';
import type { ComponentContext } from '@json-render/react';

import { catalog } from './schema';
import type {
  ChartProps,
  DataTableProps,
  FilterProps,
  DensityToggleProps,
  ViewModeToggleProps,
  TableWithFiltersProps,
  ChartWithControlsProps,
  SQLResultsDisplayProps,
} from './types';

/* ── Helper Types ────────────────────────────────────────────────────── */

/**
 * Extract the inferred type from a catalog component's props schema
 */
type InferCatalogProps<K extends keyof typeof catalog.components> =
  z.infer<(typeof catalog.components)[K]['props']>;

/* ── Type Alignment Tests ────────────────────────────────────────────── */

describe('Catalog schema types match component prop types', () => {
  it('Chart schema matches ChartProps', () => {
    type CatalogChartProps = InferCatalogProps<'Chart'>;

    // Type-level assertion: ChartProps should be assignable to catalog props
    expectTypeOf<ChartProps>().toMatchTypeOf<CatalogChartProps>();
  });

  it('DataTable schema matches DataTableProps', () => {
    type CatalogDataTableProps = InferCatalogProps<'DataTable'>;

    expectTypeOf<DataTableProps>().toMatchTypeOf<CatalogDataTableProps>();
  });

  it('Filter schema matches FilterProps', () => {
    type CatalogFilterProps = InferCatalogProps<'Filter'>;

    expectTypeOf<FilterProps>().toMatchTypeOf<CatalogFilterProps>();
  });

  it('DensityToggle schema matches DensityToggleProps', () => {
    type CatalogDensityToggleProps = InferCatalogProps<'DensityToggle'>;

    expectTypeOf<DensityToggleProps>().toMatchTypeOf<CatalogDensityToggleProps>();
  });

  it('ViewModeToggle schema matches ViewModeToggleProps', () => {
    type CatalogViewModeToggleProps = InferCatalogProps<'ViewModeToggle'>;

    expectTypeOf<ViewModeToggleProps>().toMatchTypeOf<CatalogViewModeToggleProps>();
  });

  it('TableWithFilters schema matches TableWithFiltersProps', () => {
    type CatalogTableWithFiltersProps = InferCatalogProps<'TableWithFilters'>;

    expectTypeOf<TableWithFiltersProps>().toMatchTypeOf<CatalogTableWithFiltersProps>();
  });

  it('ChartWithControls schema matches ChartWithControlsProps', () => {
    type CatalogChartWithControlsProps = InferCatalogProps<'ChartWithControls'>;

    expectTypeOf<ChartWithControlsProps>().toMatchTypeOf<CatalogChartWithControlsProps>();
  });

  it('SQLResultsDisplay schema matches SQLResultsDisplayProps', () => {
    type CatalogSQLResultsDisplayProps = InferCatalogProps<'SQLResultsDisplay'>;

    expectTypeOf<SQLResultsDisplayProps>().toMatchTypeOf<CatalogSQLResultsDisplayProps>();
  });
});

describe('Registry completeness at type level', () => {
  it('all catalog components have corresponding prop types', () => {
    // This test ensures we haven't missed any components
    type CatalogComponents = keyof typeof catalog.components;

    // If a component is added to catalog without a corresponding type,
    // this will show as a type error
    type ExpectedComponents =
      | 'Chart'
      | 'DataTable'
      | 'Filter'
      | 'DensityToggle'
      | 'ViewModeToggle'
      | 'TableWithFilters'
      | 'ChartWithControls'
      | 'SQLResultsDisplay';

    expectTypeOf<CatalogComponents>().toEqualTypeOf<ExpectedComponents>();
  });
});
```

These tests use vitest's `expectTypeOf` for compile-time type assertions. If the catalog schemas drift from the prop type interfaces, TypeScript will catch it at compile time.
