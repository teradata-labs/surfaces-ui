```tsx
import { render, screen, act } from '@testing-library/react';
import { ThemeProvider, useTheme, type DensityMode } from './ThemeProvider';

// Test component to access context
function DensityConsumer() {
  const { density, setDensity } = useTheme();
  return (
    <div>
      <span data-testid="density-value">{density}</span>
      <button onClick={() => setDensity('compact')}>Set Compact</button>
      <button onClick={() => setDensity('spacious')}>Set Spacious</button>
    </div>
  );
}

describe('ThemeProvider - Density', () => {
  // Mock localStorage
  const localStorageMock = {
    getItem: vi.fn(),
    setItem: vi.fn(),
    removeItem: vi.fn(),
    clear: vi.fn(),
  };

  beforeEach(() => {
    vi.stubGlobal('localStorage', localStorageMock);
    localStorageMock.getItem.mockClear();
    localStorageMock.setItem.mockClear();
    document.documentElement.removeAttribute('data-density');
  });

  afterEach(() => {
    vi.unstubAllGlobals();
  });

  it('defaults to comfortable density', () => {
    localStorageMock.getItem.mockReturnValue(null);
    render(
      <ThemeProvider>
        <DensityConsumer />
      </ThemeProvider>
    );
    expect(screen.getByTestId('density-value')).toHaveTextContent('comfortable');
  });

  it('accepts initial density prop', () => {
    render(
      <ThemeProvider density="compact">
        <DensityConsumer />
      </ThemeProvider>
    );
    expect(screen.getByTestId('density-value')).toHaveTextContent('compact');
  });

  it('reads density from localStorage on mount', () => {
    localStorageMock.getItem.mockReturnValue('spacious');
    render(
      <ThemeProvider>
        <DensityConsumer />
      </ThemeProvider>
    );
    expect(screen.getByTestId('density-value')).toHaveTextContent('spacious');
  });

  it('persists density changes to localStorage', () => {
    localStorageMock.getItem.mockReturnValue(null);
    render(
      <ThemeProvider>
        <DensityConsumer />
      </ThemeProvider>
    );

    act(() => {
      screen.getByText('Set Compact').click();
    });

    expect(localStorageMock.setItem).toHaveBeenCalledWith('density', 'compact');
  });

  it('applies data-density attribute to wrapper', () => {
    render(
      <ThemeProvider density="spacious">
        <DensityConsumer />
      </ThemeProvider>
    );

    // Check document element has density attribute (from effect)

  });

  it('updates data-density when density changes', () => {
    localStorageMock.getItem.mockReturnValue(null);
    render(
      <ThemeProvider>
        <DensityConsumer />
      </ThemeProvider>

    act(() => {
      screen.getByText('Set Compact').click();
    });

    expect(document.documentElement.getAttribute('data-density')).toBe('compact');
  });

  it('exposes setDensity through useTheme hook', () => {
    localStorageMock.getItem.mockReturnValue(null);
    render(
      <ThemeProvider>
        <DensityConsumer />
      </ThemeProvider>
    );

    // Initially comfortable
    expect(screen.getByTestId('density-value')).toHaveTextContent('comfortable');

    // Change to spacious

      unmount();

  });
```

```tsx
describe('Zero-Runtime Styling Constraint (THEME-06)', () => {
  it('does not import CSS-in-JS libraries', async () => {
    // Read the renderer source file
    const fs = await import('fs');
    const path = await import('path');

    const rendererPath = path.join(__dirname, 'renderer.tsx');
    const themeProviderPath = path.join(__dirname, '../theme/ThemeProvider.tsx');

    const rendererSource = fs.readFileSync(rendererPath, 'utf-8');
    const themeProviderSource = fs.readFileSync(themeProviderPath, 'utf-8');

    // CSS-in-JS libraries that would violate zero-runtime
    const cssInJsPatterns = [
      'styled-components',
      '@emotion',
      'linaria',
      'vanilla-extract',
      'stitches',
      'goober',
    ];

    cssInJsPatterns.forEach(pattern => {
      expect(rendererSource).not.toContain(pattern);
      expect(themeProviderSource).not.toContain(pattern);
    });
  });

  it('uses CSS custom properties for theming', async () => {
    const fs = await import('fs');
    const path = await import('path');

    const tokensCssPath = path.join(__dirname, '../theme/tokens.css');
    const densityCssPath = path.join(__dirname, '../theme/density.css');

    const tokensCss = fs.readFileSync(tokensCssPath, 'utf-8');
    const densityCss = fs.readFileSync(densityCssPath, 'utf-8');

    // Verify CSS custom properties are used
    expect(tokensCss).toContain('--');
    expect(densityCss).toContain('--density-scale');
    expect(densityCss).toContain('data-density');
  });
});
```
