Enhance ThemeProvider with density state management:

1. Add DensityMode type:
   ```tsx
   export type DensityMode = 'compact' | 'comfortable' | 'spacious';
   ```

2. Update ThemeConfig interface:
   ```tsx
   export interface ThemeConfig {
     theme: ThemeMode;
     setTheme: (mode: ThemeMode) => void;
     resolvedTheme: 'light' | 'dark';
     density: DensityMode;        // NEW
     setDensity: (mode: DensityMode) => void;  // NEW
   }
   ```

3. Update ThemeProviderProps:
   ```tsx
   interface ThemeProviderProps {
     theme?: ThemeMode;
     density?: DensityMode;  // NEW - defaults to 'comfortable'
     tokenOverrides?: Record<string, string>;
     children: React.ReactNode;
   }
   ```

4. Add density state in component:
   ```tsx
   const [density, setDensityState] = useState<DensityMode>(
     initialDensity ?? 'comfortable'
   );

   const setDensity = useCallback((mode: DensityMode) => {
     setDensityState(mode);
   }, []);
   ```

5. Add density persistence effect:
   ```tsx
   useEffect(() => {
     localStorage.setItem('density', density);
     document.documentElement.setAttribute('data-density', density);
   }, [density]);
   ```

6. Add density to context value:
   ```tsx
   const value = useMemo<ThemeConfig>(
     () => ({ theme, setTheme, resolvedTheme, density, setDensity }),
     [theme, setTheme, resolvedTheme, density, setDensity]
   );
   ```

7. Update wrapper div to ALWAYS render (needed for data-density):
   ```tsx
   return (
     <ThemeContext.Provider value={value}>
       <div data-density={density} style={overrideStyle}>
         {children}
       </div>
     </ThemeContext.Provider>
   );
   ```

Note: The wrapper div now always renders to apply data-density attribute. This is a change from the conditional rendering but necessary for density to work via CSS inheritance.

8. Add density initialization from localStorage in initial state:
   ```tsx
   const [density, setDensityState] = useState<DensityMode>(() => {
     if (initialDensity) return initialDensity;
     if (typeof window !== 'undefined') {
       const stored = localStorage.getItem('density');
       if (stored === 'compact' || stored === 'comfortable' || stored === 'spacious') {
         return stored;
       }
     }
     return 'comfortable';
   });
   ```
