```typescript
import { describe, it, expect } from 'vitest';
import { validateSpec } from './validation';

describe('Spec Validation', () => {
  describe('Valid Specs', () => {
    it('validates DataTable spec with all required props', () => {
      const spec = {
        type: 'DataTable',
        props: { columns: ['id', 'name'] },
      };
      const result = validateSpec(spec);
      expect(result.valid).toBe(true);
      expect(result.errors).toHaveLength(0);
    });

    it('validates nested molecule spec (TableWithFilters)', () => {
      // TEST-02: Registry mapping validation
      const spec = {
        type: 'TableWithFilters',
        props: {
          columns: ['id', 'name'],
          filterColumns: ['name'],
        },
      };
      const result = validateSpec(spec);
      expect(result.valid).toBe(true);
    });

    it('validates organism spec (SQLResultsDisplay)', () => {
      const spec = {
        type: 'SQLResultsDisplay',
        props: { initialViewMode: 'table' },
      };
      const result = validateSpec(spec);
      expect(result.valid).toBe(true);
    });
  });

  describe('Invalid Specs', () => {
    it('rejects spec with unknown component type', () => {
      const spec = { type: 'UnknownComponent', props: {} };
      const result = validateSpec(spec);
      expect(result.valid).toBe(false);
      expect(result.errors).toContain('Unknown component type');
    });

    it('rejects spec with invalid prop types', () => {
      const spec = {
        type: 'DataTable',
        props: { columns: 'not-an-array' }, // Should be string[]
      };
      const result = validateSpec(spec);
      expect(result.valid).toBe(false);
    });

    it('rejects spec missing required props', () => {
      const spec = { type: 'DataTable', props: {} }; // Missing columns
      const result = validateSpec(spec);
      expect(result.valid).toBe(false);
    });

    it('rejects malformed JSON structure', () => {
      const spec = { notType: 'foo', notProps: {} }; // Wrong shape
      const result = validateSpec(spec);
      expect(result.valid).toBe(false);
    });
  });

  describe('Error Messages', () => {
    it('provides actionable error for missing required prop', () => {
      const spec = { type: 'Chart', props: {} }; // Missing chartType
      const result = validateSpec(spec);
      expect(result.errors[0]).toMatch(/chartType.*required/i);
    });

    it('provides actionable error for wrong prop type', () => {
      const spec = {
        type: 'DataTable',
        props: { columns: 123 }, // Should be array
      };
      const result = validateSpec(spec);
      expect(result.errors[0]).toMatch(/columns.*array/i);
    });
  });
});
```

```typescript
import { describe, it, expect } from 'vitest';
import { generateCatalogPrompt, generateComponentExamples } from './prompt';

describe('Prompt Generation', () => {
  describe('Catalog Prompt', () => {
    it('generates prompt with all component types', () => {
      const prompt = generateCatalogPrompt();

      // Should include organisms section
      expect(prompt).toContain('Organisms');
      expect(prompt).toContain('SQLResultsDisplay');

      // Should include molecules section
      expect(prompt).toContain('Molecules');
      expect(prompt).toContain('TableWithFilters');
      expect(prompt).toContain('ChartWithControls');

      // Should include primitives section
      expect(prompt).toContain('Primitives');
      expect(prompt).toContain('DataTable');
      expect(prompt).toContain('Chart');
      expect(prompt).toContain('Filter');
    });

    it('includes prop descriptions for each component', () => {
      const prompt = generateCatalogPrompt();

      // DataTable props described
      expect(prompt).toMatch(/columns.*string\[\]/i);
      expect(prompt).toMatch(/sortable.*boolean/i);

      // Chart props described
      expect(prompt).toMatch(/chartType.*bar|line|pie/i);
      expect(prompt).toMatch(/xAxis.*string/i);
    });

    it('lists organisms first as recommended', () => {
      // Per Phase 02 decision: Organisms listed first
      const prompt = generateCatalogPrompt();
      const organismsIndex = prompt.indexOf('Organisms');
      const moleculesIndex = prompt.indexOf('Molecules');
      const primitivesIndex = prompt.indexOf('Primitives');

      expect(organismsIndex).toBeLessThan(moleculesIndex);
      expect(moleculesIndex).toBeLessThan(primitivesIndex);
    });

    it('keeps descriptions under 60 chars per Phase 02 decision', () => {
      // Per Phase 02-01: Schema descriptions under 60 chars for AI processing
      const prompt = generateCatalogPrompt();
      const descriptionLines = prompt.split('\n').filter(line =>
        line.includes('description:')
      );

      descriptionLines.forEach(line => {
        const description = line.split('description:')[1]?.trim();
        expect(description?.length || 0).toBeLessThanOrEqual(60);
      });
    });
  });

  describe('Component Examples', () => {
    it('generates example prompts for primitives', () => {
      // DOC-02: Example prompts for each component
      const examples = generateComponentExamples('DataTable');

      expect(examples).toHaveLength(3); // 3-5 examples per component
      expect(examples[0]).toMatch(/show.*table/i);
      expect(examples[0]).toContain('DataTable');
    });

    it('generates example prompts for molecules', () => {
      const examples = generateComponentExamples('TableWithFilters');

      expect(examples).toHaveLength(3);
      expect(examples[0]).toMatch(/table.*filter/i);
    });

    it('generates example prompts for organism', () => {
      const examples = generateComponentExamples('SQLResultsDisplay');

      expect(examples).toHaveLength(3);
      expect(examples[0]).toMatch(/sql.*results/i);
    });

    it('example prompts are natural language (not technical)', () => {
      const examples = generateComponentExamples('Chart');

      examples.forEach(example => {
        // Should be conversational, not JSON or technical jargon
        expect(example).not.toMatch(/\{|\}|\[|\]/); // No JSON syntax
        expect(example).not.toMatch(/type:|props:/); // No object literals
        expect(example.split(' ').length).toBeGreaterThan(3); // Multi-word
      });
    });
  });
});
```
