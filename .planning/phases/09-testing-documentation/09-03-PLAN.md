**Test structure** (following Phase 07 patterns):
```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { renderHook, waitFor } from '@testing-library/react';
import { useSpecStream } from './stream';

// Mock fetch for streaming JSONL responses
global.fetch = vi.fn();

describe('useSpecStream Hook', () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe('Successful Streaming', () => {
    it('streams spec chunks progressively', async () => {
      // TEST-03: NL generation flow (prompt → spec → rendered UI)
      const mockStream = new ReadableStream({
        start(controller) {
          controller.enqueue(
            new TextEncoder().encode('{"chunk": "{\\"type\\": "}\n')
          );
          controller.enqueue(
            new TextEncoder().encode('{"chunk": "\\"DataTable\\""}\n')
          );
          controller.enqueue(
            new TextEncoder().encode('{"chunk": ", \\"props\\": {}}"}\n')
          );
          controller.close();
        },
      });

      (global.fetch as any).mockResolvedValue({
        ok: true,
        body: mockStream,
      });

      const { result } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'show table' })
      );

      await waitFor(() => {
        expect(result.current.spec).toBeTruthy();
      });

      // Verify progressive updates occurred
      expect(result.current.spec).toContain('DataTable');
      expect(result.current.progress).toBe(100);
      expect(result.current.error).toBeNull();
    });

    it('updates progress based on chunk accumulation', async () => {
      const progressUpdates: number[] = [];
      const mockStream = new ReadableStream({
        start(controller) {
          controller.enqueue(new TextEncoder().encode('{"chunk": "{"}\n'));
          setTimeout(() => {
            controller.enqueue(
              new TextEncoder().encode('{"chunk": "\\"type\\""}\n')
            );
            setTimeout(() => controller.close(), 100);
          }, 100);
        },
      });

      (global.fetch as any).mockResolvedValue({
        ok: true,
        body: mockStream,
      });

      const { result } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'test' })
      );

      // Capture progress at different intervals
      await waitFor(() => expect(result.current.progress).toBeGreaterThan(0));
      progressUpdates.push(result.current.progress);

      await waitFor(() => expect(result.current.progress).toBe(100));
      progressUpdates.push(result.current.progress);

      // Progress should increase over time
      expect(progressUpdates[0]).toBeLessThan(progressUpdates[1]);
    });

    it('handles explanation chunks before spec', async () => {
      // Phase 07-03: Dual-call approach with explanation first
      const mockStream = new ReadableStream({
        start(controller) {
          controller.enqueue(
            new TextEncoder().encode(
              '{"explanation": "I chose DataTable because..."}\n'
            )
          );
          controller.enqueue(
            new TextEncoder().encode(
              '{"chunk": "{\\"type\\": \\"DataTable\\"}"}\n'
            )
          );
          controller.close();
        },
      });

      (global.fetch as any).mockResolvedValue({
        ok: true,
        body: mockStream,
      });

      const { result } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'test' })
      );

      await waitFor(() => {
        expect(result.current.explanation).toBeTruthy();
      });

      expect(result.current.explanation).toContain('I chose DataTable');
    });
  });

  describe('Error Handling', () => {
    it('handles network errors gracefully', async () => {
      (global.fetch as any).mockRejectedValue(new Error('Network error'));

      const { result } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'test' })
      );

      await waitFor(() => {
        expect(result.current.error).toBeTruthy();
      });

      expect(result.current.error).toContain('Network error');
      expect(result.current.loading).toBe(false);
    });

    it('handles HTTP error responses', async () => {
      (global.fetch as any).mockResolvedValue({
        ok: false,
        status: 500,
        statusText: 'Internal Server Error',
      });

      const { result } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'test' })
      );

      await waitFor(() => {
        expect(result.current.error).toBeTruthy();
      });

      expect(result.current.error).toContain('500');
    });

    it('handles malformed JSONL chunks', async () => {
      const mockStream = new ReadableStream({
        start(controller) {
          controller.enqueue(new TextEncoder().encode('not-valid-json\n'));
          controller.close();
        },
      });

      (global.fetch as any).mockResolvedValue({
        ok: true,
        body: mockStream,
      });

      const { result } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'test' })
      );

      await waitFor(() => {
        expect(result.current.error).toBeTruthy();
      });
    });
  });

  describe('Cancellation', () => {
    it('cancels stream via AbortController', async () => {
      const mockAbort = vi.fn();
      const mockStream = new ReadableStream({
        start(controller) {
          setTimeout(() => controller.close(), 1000); // Long stream
        },
      });

      (global.fetch as any).mockImplementation((url, options) => {
        options?.signal?.addEventListener('abort', mockAbort);
        return Promise.resolve({ ok: true, body: mockStream });
      });

      const { result, unmount } = renderHook(() =>
        useSpecStream('/api/generate-ui', { prompt: 'test' })
      );

      // Start streaming
      await waitFor(() => expect(result.current.loading).toBe(true));

      // Cancel by unmounting
      unmount();

      // Verify AbortController called
      await waitFor(() => expect(mockAbort).toHaveBeenCalled());
    });
  });
});
```

import { describe, it, expect, vi } from 'vitest';
import { render, screen } from '@testing-library/react';
import { axe, toHaveNoViolations } from 'jest-axe';
import userEvent from '@testing-library/user-event';
import { NLGenerator } from './NLGenerator';

expect.extend(toHaveNoViolations);

describe('NLGenerator Accessibility', () => {
  describe('axe-core Compliance', () => {
    it('has no accessibility violations in default state', async () => {
      // TEST-05: Accessibility tests with axe-core
      const { container } = render(
        <NLGenerator
          baseUrl="http://localhost:8000"
          showExamples={true}
        />
      );

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });

    it('has no violations with example prompts visible', async () => {
      const { container } = render(
        <NLGenerator
          baseUrl="http://localhost:8000"
          showExamples={true}
        />
      );

      // Example buttons should be present
      const exampleButton = screen.getAllByRole('button')[0];
      expect(exampleButton).toBeInTheDocument();

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });

    it('has no violations during loading state', async () => {
      const { container } = render(
        <NLGenerator
          baseUrl="http://localhost:8000"
        />
      );

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });

    it('has no violations with error message displayed', async () => {
      const { container } = render(
        <NLGenerator
          baseUrl="http://localhost:8000"
        />
      );

      // Trigger error state (implementation depends on component)
      // Mock fetch to return error
      global.fetch = vi.fn().mockRejectedValue(new Error('API error'));

      const user = userEvent.setup();
      const input = screen.getByPlaceholderText(/describe/i);
      await user.type(input, 'test prompt');

      const generateBtn = screen.getByRole('button', { name: /generate/i });
      await user.click(generateBtn);

      // Wait for error to appear
      await screen.findByRole('alert');

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });
  });

  describe('Keyboard Navigation', () => {
    it('focuses prompt input on Tab', async () => {
      render(<NLGenerator baseUrl="http://localhost:8000" />);

      const user = userEvent.setup();
      await user.tab();

      const input = screen.getByPlaceholderText(/describe/i);
      expect(input).toHaveFocus();
    });

    it('submits on Enter key in input', async () => {
      const mockRenderSpec = vi.fn();
      render(
        <NLGenerator
          baseUrl="http://localhost:8000"
          renderSpec={mockRenderSpec}
        />
      );

      const user = userEvent.setup();
      const input = screen.getByPlaceholderText(/describe/i);

      await user.type(input, 'show table');
      await user.keyboard('{Enter}');

      // Verify generation triggered
      expect(mockRenderSpec).toHaveBeenCalled();
    });

    it('navigates example buttons with Tab', async () => {
      render(
        <NLGenerator
          baseUrl="http://localhost:8000"
          showExamples={true}
        />
      );

      const user = userEvent.setup();
      const exampleButtons = screen.getAllByRole('button').filter(btn =>
        btn.textContent?.includes('Show') || btn.textContent?.includes('Display')
      );

      // Tab through example buttons
      await user.tab(); // Input
      await user.tab(); // Generate button
      await user.tab(); // First example
      expect(exampleButtons[0]).toHaveFocus();

      await user.tab(); // Second example
      expect(exampleButtons[1]).toHaveFocus();
    });
  });

  describe('ARIA Attributes', () => {
    it('has accessible name for prompt input', () => {
      render(<NLGenerator baseUrl="http://localhost:8000" />);

      const input = screen.getByPlaceholderText(/describe/i);
      expect(input).toHaveAttribute('aria-label');
      expect(input.getAttribute('aria-label')).toBeTruthy();
    });

    it('generates button has accessible name', () => {
      render(<NLGenerator baseUrl="http://localhost:8000" />);

      const generateBtn = screen.getByRole('button', { name: /generate/i });
      expect(generateBtn).toBeInTheDocument();
    });

    it('error messages have role="alert"', async () => {
      global.fetch = vi.fn().mockRejectedValue(new Error('API error'));

      render(<NLGenerator baseUrl="http://localhost:8000" />);

      const user = userEvent.setup();
      const input = screen.getByPlaceholderText(/describe/i);
      await user.type(input, 'test');

      const generateBtn = screen.getByRole('button', { name: /generate/i });
      await user.click(generateBtn);

      const alert = await screen.findByRole('alert');
      expect(alert).toBeInTheDocument();
    });

    it('loading state announced to screen readers', async () => {
      render(<NLGenerator baseUrl="http://localhost:8000" />);

      // Check for aria-live region or loading indicator
      const loadingIndicator = screen.queryByRole('status');
      // If no status role, check for aria-busy or aria-live
      if (!loadingIndicator) {
        const container = screen.getByTestId('nl-generator');
        expect(
          container.hasAttribute('aria-busy') ||
          container.hasAttribute('aria-live')
        ).toBe(true);
      }
    });
  });
});

import { describe, it, expect, vi, beforeEach } from 'vitest';
import { render, screen } from '@testing-library/react';
import { axe, toHaveNoViolations } from 'jest-axe';
import { Chart } from './Chart';

expect.extend(toHaveNoViolations);

// Mock ECharts to avoid canvas rendering issues in jsdom
vi.mock('echarts', () => ({
  init: vi.fn(() => ({
    setOption: vi.fn(),
    resize: vi.fn(),
    dispose: vi.fn(),
    showLoading: vi.fn(),
    hideLoading: vi.fn(),
  })),
  registerTheme: vi.fn(),
}));

describe('Chart Accessibility', () => {
  const mockData = [
    { category: 'A', value: 10 },
    { category: 'B', value: 20 },
  ];

  beforeEach(() => {
    vi.clearAllMocks();
  });

  describe('axe-core Compliance', () => {
    it('has no accessibility violations', async () => {
      const { container } = render(
        <Chart
          data={mockData}
          chartType="bar"
          xAxis="category"
          yAxis="value"
        />
      );

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });

    it('has no violations with chart title', async () => {
      const { container } = render(
        <Chart
          data={mockData}
          chartType="line"
          xAxis="category"
          yAxis="value"
          title="Sales Over Time"
        />
      );

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });
  });

  describe('ARIA Attributes', () => {
    it('has role="img" for chart container', () => {
      render(
        <Chart
          data={mockData}
          chartType="bar"
          xAxis="category"
          yAxis="value"
        />
      );

      const chartContainer = screen.getByRole('img');
      expect(chartContainer).toBeInTheDocument();
    });

    it('has aria-label describing chart type', () => {
      render(
        <Chart
          data={mockData}
          chartType="pie"
          xAxis="category"
          yAxis="value"
        />
      );

      const chartContainer = screen.getByRole('img');
      expect(chartContainer).toHaveAttribute('aria-label');
      expect(chartContainer.getAttribute('aria-label')).toMatch(/pie chart/i);
    });

    it('includes data summary in aria-label', () => {
      render(
        <Chart
          data={mockData}
          chartType="bar"
          xAxis="category"
          yAxis="value"
        />
      );

      const chartContainer = screen.getByRole('img');
      const label = chartContainer.getAttribute('aria-label');

      // Should mention data points count
      expect(label).toContain('2');
    });

    it('uses title in aria-label when provided', () => {
      render(
        <Chart
          data={mockData}
          chartType="line"
          xAxis="category"
          yAxis="value"
          title="Revenue Trends"
        />
      );

      const chartContainer = screen.getByRole('img');
      expect(chartContainer.getAttribute('aria-label')).toContain('Revenue Trends');
    });
  });

  describe('Screen Reader Support', () => {
    it('provides text alternative for visual data', () => {
      render(
        <Chart
          data={mockData}
          chartType="bar"
          xAxis="category"
          yAxis="value"
        />
      );

      // Chart should have accessible description
      const chartContainer = screen.getByRole('img');
      expect(
        chartContainer.getAttribute('aria-label') ||
        chartContainer.getAttribute('aria-describedby')
      ).toBeTruthy();
    });

    it('announces loading state to screen readers', () => {
      render(
        <Chart
          data={mockData}
          chartType="bar"
          xAxis="category"
          yAxis="value"
          loading={true}
        />
      );

      // Loading indicator should have role="status" or aria-live
      const loadingElement = screen.queryByRole('status');
      if (!loadingElement) {
        const container = screen.getByRole('img');
        expect(container.hasAttribute('aria-busy')).toBe(true);
      }
    });
  });

  describe('Color and Contrast', () => {
    it('does not rely on color alone for information', async () => {
      // This is primarily a visual check but we can verify structure
      const { container } = render(
        <Chart
          data={mockData}
          chartType="bar"
          xAxis="category"
          yAxis="value"
        />
      );

      // Verify chart has accessible label (text alternative to color)
      const chartContainer = screen.getByRole('img');
      expect(chartContainer.getAttribute('aria-label')).toBeTruthy();

      const results = await axe(container);
      expect(results).toHaveNoViolations();
    });
  });
});
