```typescript
/**
 * Chart Utility Functions
 *
 * Extracted from SQLResultsDisplay for reuse in Chart primitive.
 */
import type * as echarts from 'echarts';
import type { ChartType, ChartConfig, AggregationType } from './types';
import { DEFAULT_CHART_COLORS } from './types';

/**
 * Detect the system theme (dark/light) for ECharts
 */
export function getSystemTheme(): string {
  if (typeof window !== 'undefined') {
    if (
      document.documentElement.classList.contains('dark') ||
      document.body.classList.contains('dark')
    ) {
      return 'professionalDark';
    }
    if (
      window.matchMedia &&
      window.matchMedia('(prefers-color-scheme: dark)').matches
    ) {
      return 'professionalDark';
    }
  }
  return 'professionalLight';
}

/**
 * Auto-detect best axes from data structure
 */
export function detectBestAxes(
  rows: Record<string, unknown>[],
  columns: string[]
): { xAxis: string; yAxis: string } {
  if (!rows.length || !columns.length) return { xAxis: '', yAxis: '' };

  const firstRow = rows[0];
  if (!firstRow) return { xAxis: columns[0] || '', yAxis: columns[1] || columns[0] || '' };

  const categoricalCandidates = columns.filter((col) => {
    const sampleValue = firstRow[col];
    return typeof sampleValue === 'string';
  });

  const numericalCandidates = columns.filter((col) => {
    const sampleValue = firstRow[col];
    const numValue = parseFloat(String(sampleValue));
    return !isNaN(numValue) && isFinite(numValue);
  });

  const xAxis = categoricalCandidates[0] || columns[0] || '';
  const yAxis =
    numericalCandidates.find((col) => col !== xAxis) ||
    numericalCandidates[0] ||
    columns[1] ||
    columns[0] ||
    '';

  return { xAxis, yAxis };
}

/**
 * Generate ECharts option for a simple chart (no grouping/stacking)
 * For advanced configs, use generateAdvancedChartOption
 */
export function generateChartOption(
  data: Record<string, unknown>[],
  xAxis: string,
  yAxis: string,
  type: ChartType = 'bar',
  showDataLabels = false,
  colors: string[] = DEFAULT_CHART_COLORS
): echarts.EChartsOption {
  if (!data.length) return {};

  const xData = data.map((row) => String(row[xAxis] || ''));
  const yData = data.map((row) => {
    const val = parseFloat(String(row[yAxis]));
    return isNaN(val) ? 0 : val;
  });

  const categoryAxisStyle = {
    type: 'category' as const,
    data: xData,
    axisLabel: {
      interval: 0,
      fontSize: 12,
      color: '#6b7280',
      rotate: xData.some((label) => label.length > 8) ? 45 : 0,
    },
    axisLine: { lineStyle: { color: '#6b7280', type: 'dotted' as const } },
    axisTick: { lineStyle: { color: '#6b7280' } },
  };

  const valueAxisStyle = {
    type: 'value' as const,
    axisLabel: { fontSize: 12, color: '#6b7280' },
    axisLine: { lineStyle: { color: '#6b7280', type: 'dotted' as const } },
    axisTick: { lineStyle: { color: '#6b7280' } },
    splitLine: { lineStyle: { color: '#6b7280', type: 'dotted' as const, opacity: 0.3 } },
  };

  const baseOption: echarts.EChartsOption = {
    tooltip: {
      trigger: type === 'pie' ? 'item' : 'axis',
      axisPointer: type !== 'pie' ? { type: 'shadow' } : undefined,
    },
    grid: type !== 'pie' ? {
      left: '3%', right: '4%', bottom: '3%', top: '10%', containLabel: true,
    } : undefined,
  };

  // Type-specific series configuration
  switch (type) {
    case 'pie':
      return {
        ...baseOption,
        series: [{
          name: yAxis,
          type: 'pie',
          radius: '60%',
          data: xData.map((name, idx) => ({ name, value: yData[idx] })),
          emphasis: { itemStyle: { shadowBlur: 10, shadowOffsetX: 0, shadowColor: 'rgba(0, 0, 0, 0.5)' } },
          label: showDataLabels ? { show: true } : { show: false },
        }],
      };

    case 'scatter':
      return {
        ...baseOption,
        xAxis: { ...valueAxisStyle, name: xAxis },
        yAxis: { ...valueAxisStyle, name: yAxis },
        series: [{
          type: 'scatter',
          data: data.map((row) => [
            parseFloat(String(row[xAxis])) || 0,
            parseFloat(String(row[yAxis])) || 0,
          ]),
          symbolSize: 8,
          itemStyle: { color: colors[0] },
        }],
      };

    case 'heatmap':
      return {
        ...baseOption,
        xAxis: { ...categoryAxisStyle },
        yAxis: { type: 'category', data: ['Value'] },
        visualMap: {
          min: Math.min(...yData),
          max: Math.max(...yData),
          calculable: true,
          orient: 'horizontal',
          left: 'center',
          bottom: '0%',
        },
        series: [{
          type: 'heatmap',
          data: yData.map((val, idx) => [idx, 0, val]),
          label: showDataLabels ? { show: true } : { show: false },
        }],
      };

    case 'area':
      return {
        ...baseOption,
        xAxis: { ...categoryAxisStyle },
        yAxis: { ...valueAxisStyle },
        series: [{
          name: yAxis,
          type: 'line',
          data: yData,
          areaStyle: {},
          smooth: true,
          itemStyle: { color: colors[0] },
        }],
      };

    case 'line':
      return {
        ...baseOption,
        xAxis: { ...categoryAxisStyle },
        yAxis: { ...valueAxisStyle },
        series: [{
          name: yAxis,
          type: 'line',
          data: yData,
          smooth: true,
          itemStyle: { color: colors[0] },
        }],
      };

    case 'bar':
    default:
      return {
        ...baseOption,
        xAxis: { ...categoryAxisStyle },
        yAxis: { ...valueAxisStyle },
        series: [{
          name: yAxis,
          type: 'bar',
          data: yData,
          itemStyle: { color: colors[0] },
          barWidth: '60%',
          label: showDataLabels ? { show: true, position: 'top' as const } : undefined,
        }],
      };
  }
}
```

```typescript
/**
 * Chart Primitive
 *
 * Standalone ECharts wrapper with theme integration.
 * Supports 8 chart types with automatic theme detection.
 *
 * Props limited to 8 for AI describability (json-render catalog constraint).
 */
import * as React from 'react';
import * as echarts from 'echarts';
import { registerEChartsThemes } from '../../components/charts/echart-themes';
import { cn } from '../../lib/utils';
import { generateChartOption, getSystemTheme, detectBestAxes } from './utils';
import type { ChartProps, ChartType } from './types';

// Register themes on module load (idempotent)
registerEChartsThemes(echarts);

export function Chart({
  data,
  xAxis: xAxisProp,
  yAxis: yAxisProp,
  type = 'bar',
  theme: themeProp,
  height = 400,
  showDataLabels = false,
  className,
}: ChartProps) {
  const chartRef = React.useRef<HTMLDivElement>(null);
  const chartInstance = React.useRef<echarts.ECharts | null>(null);

  // Determine axes - use provided or auto-detect
  const columns = React.useMemo(() => {
    if (!data.length) return [];
    return Object.keys(data[0] || {});
  }, [data]);

  const { xAxis, yAxis } = React.useMemo(() => {
    if (xAxisProp && yAxisProp) {
      return { xAxis: xAxisProp, yAxis: yAxisProp };
    }
    const detected = detectBestAxes(data, columns);
    return {
      xAxis: xAxisProp || detected.xAxis,
      yAxis: yAxisProp || detected.yAxis,
    };
  }, [xAxisProp, yAxisProp, data, columns]);

  // Determine theme - use provided or detect system theme
  const theme = React.useMemo(() => {
    if (themeProp) return themeProp;
    return getSystemTheme();
  }, [themeProp]);

  // Initialize chart with theme
  React.useEffect(() => {
    if (!chartRef.current) return;

    // Dispose existing instance (theme change requires re-init)
    if (chartInstance.current) {
      chartInstance.current.dispose();
    }

    chartInstance.current = echarts.init(chartRef.current, theme);

    return () => {
      chartInstance.current?.dispose();
      chartInstance.current = null;
    };
  }, [theme]);

  // Update chart options when data or config changes
  React.useEffect(() => {
    if (!chartInstance.current || !data.length) return;

    const option = generateChartOption(data, xAxis, yAxis, type, showDataLabels);
    chartInstance.current.setOption(option, { notMerge: true });
  }, [data, xAxis, yAxis, type, showDataLabels]);

  // Handle resize
  React.useEffect(() => {
    const handleResize = () => chartInstance.current?.resize();
    window.addEventListener('resize', handleResize);

    // ResizeObserver for container changes
    let observer: ResizeObserver | null = null;
    if (chartRef.current && typeof ResizeObserver !== 'undefined') {
      observer = new ResizeObserver(() => {
        setTimeout(() => chartInstance.current?.resize(), 100);
      });
      observer.observe(chartRef.current);
    }

    return () => {
      window.removeEventListener('resize', handleResize);
      observer?.disconnect();
    };
  }, []);

  if (!data.length) {
    return (
      <div
        className={cn('flex items-center justify-center text-muted-foreground', className)}
        style={{ height: `${height}px` }}
      >
        No data to display
      </div>
    );
  }

  return (
    <div
      ref={chartRef}
      className={cn('w-full', className)}
      style={{ height: `${height}px` }}
      role="img"
      aria-label={`${type} chart showing ${yAxis} by ${xAxis}`}
    />
  );
}

export default Chart;
```
