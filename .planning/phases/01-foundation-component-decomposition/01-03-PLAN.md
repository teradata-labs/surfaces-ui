```typescript
/**
 * VirtualRow - Memoized row component for DataTable
 *
 * Extracted as separate component to enable React.memo optimization.
 * Critical for 60FPS scrolling with large datasets.
 */
import * as React from 'react';
import { cn } from '../../lib/utils';
import type { VirtualRowProps, Density } from './types';
import { ROW_HEIGHTS } from './types';

function VirtualRowComponent({
  row,
  columns,
  index,
  style,
  density,
  onClick,
  showRowNumbers,
}: VirtualRowProps) {
  const handleClick = React.useCallback(() => {
    onClick?.(row, index);
  }, [onClick, row, index]);

  const rowHeight = ROW_HEIGHTS[density];
  const paddingClass = density === 'compact' ? 'px-2 py-1' : density === 'spacious' ? 'px-4 py-3' : 'px-3 py-2';

  return (
    <tr
      className={cn(
        'border-b border-border last:border-b-0 hover:bg-muted/30',
        onClick && 'cursor-pointer'
      )}
      style={{ ...style, height: `${rowHeight}px` }}
      onClick={onClick ? handleClick : undefined}
      role="row"
    >
      {showRowNumbers && (
        <td className={cn(paddingClass, 'text-muted-foreground text-right w-12')}>
          {index + 1}
        </td>
      )}
      {columns.map((col) => (
        <td
          key={col}
          className={cn(paddingClass, 'text-foreground leading-tight max-w-xs truncate')}
          title={String(row[col] ?? '')}
        >
          {row[col] != null ? (
            String(row[col])
          ) : (
            <span className="text-muted-foreground italic">null</span>
          )}
        </td>
      ))}
    </tr>
  );
}

// Memoize to prevent unnecessary re-renders during virtual scrolling
export const VirtualRow = React.memo(VirtualRowComponent);
```

```typescript
/**
 * DataTable Primitive
 *
 * Virtual scrolling data table using TanStack Virtual.
 * Handles 100K+ rows at 60FPS with memoized row components.
 *
 * Props limited to 8 for AI describability (json-render catalog constraint).
 */
import * as React from 'react';
import { useVirtualizer } from '@tanstack/react-virtual';
import { cn } from '../../lib/utils';
import { VirtualRow } from './VirtualRow';
import type { DataTableProps, Density } from './types';
import { ROW_HEIGHTS } from './types';

const OVERSCAN = 5; // Extra rows rendered above/below viewport

export function DataTable({
  data,
  columns: columnsProp,
  height = 400,
  density = 'comfortable',
  virtualScrolling = true,
  onRowClick,
  className,
  showRowNumbers = false,
}: DataTableProps) {
  const parentRef = React.useRef<HTMLDivElement>(null);

  // Derive columns from data if not provided
  const columns = React.useMemo(() => {
    if (columnsProp?.length) return columnsProp;
    if (!data.length) return [];
    return Object.keys(data[0] || {});
  }, [columnsProp, data]);

  const rowHeight = ROW_HEIGHTS[density];

  // Virtual scrolling setup
  const rowVirtualizer = useVirtualizer({
    count: data.length,
    getScrollElement: () => parentRef.current,
    estimateSize: () => rowHeight,
    overscan: OVERSCAN,
  });

  const virtualItems = rowVirtualizer.getVirtualItems();
  const totalSize = rowVirtualizer.getTotalSize();

  // Empty state
  if (!data.length) {
    return (
      <div
        className={cn(
          'flex items-center justify-center text-muted-foreground border border-border rounded-lg',
          className
        )}
        style={{ height: `${height}px` }}
      >
        No data to display
      </div>
    );
  }

  // Non-virtual rendering for small datasets
  if (!virtualScrolling || data.length <= 100) {
    return (
      <div
        className={cn('border border-border rounded-lg overflow-hidden', className)}
      >
        <div className="overflow-auto" style={{ maxHeight: `${height}px` }}>
          <table className="w-full text-xs" role="grid">
            <thead className="bg-muted sticky top-0 z-10">
              <tr>
                {showRowNumbers && (
                  <th className="px-3 py-1.5 text-right font-medium text-muted-foreground border-b border-border bg-muted w-12">
                    #
                  </th>
                )}
                {columns.map((col) => (
                  <th
                    key={col}
                    className="px-3 py-1.5 text-left font-medium text-muted-foreground border-b border-border bg-muted"
                  >
                    {col}
                  </th>
                ))}
              </tr>
            </thead>
            <tbody>
              {data.map((row, idx) => (
                <VirtualRow
                  key={idx}
                  row={row}
                  columns={columns}
                  index={idx}
                  style={{}}
                  density={density}
                  onClick={onRowClick}
                  showRowNumbers={showRowNumbers}
                />
              ))}
            </tbody>
          </table>
        </div>
      </div>
    );
  }

  // Virtual scrolling for large datasets
  return (
    <div
      className={cn('border border-border rounded-lg overflow-hidden', className)}
    >
      <div
        ref={parentRef}
        className="overflow-auto scrollbar-thin"
        style={{ height: `${height}px` }}
      >
        <table className="w-full text-xs" role="grid">
          <thead className="bg-muted sticky top-0 z-10">
            <tr>
              {showRowNumbers && (
                <th className="px-3 py-1.5 text-right font-medium text-muted-foreground border-b border-border bg-muted w-12">
                  #
                </th>
              )}
              {columns.map((col) => (
                <th
                  key={col}
                  className="px-3 py-1.5 text-left font-medium text-muted-foreground border-b border-border bg-muted"
                >
                  {col}
                </th>
              ))}
            </tr>
          </thead>
          <tbody style={{ height: `${totalSize}px`, position: 'relative' }}>
            {virtualItems.map((virtualRow) => (
              <VirtualRow
                key={virtualRow.index}
                row={data[virtualRow.index]}
                columns={columns}
                index={virtualRow.index}
                style={{
                  position: 'absolute',
                  top: 0,
                  left: 0,
                  width: '100%',
                  transform: `translateY(${virtualRow.start}px)`,
                }}
                density={density}
                onClick={onRowClick}
                showRowNumbers={showRowNumbers}
              />
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

export default DataTable;
```
