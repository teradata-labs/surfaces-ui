```typescript
import { Copy, Download, ExternalLink } from 'lucide-react';
import type { SQLResultsAction } from './types';

/**
 * Create default actions for the results toolbar
 */
export function createDefaultActions(options: {
  projectId?: string;
  teradataBaseUrl?: string;
}): SQLResultsAction[] {
  const actions: SQLResultsAction[] = [
    {
      id: 'copy',
      label: 'Copy',
      icon: Copy,
      tooltip: 'Copy as JSON',
      onClick: async ({ data }) => {
        try {
          await navigator.clipboard.writeText(JSON.stringify(data, null, 2));
        } catch {
          // Fallback for older browsers
          const textarea = document.createElement('textarea');
          textarea.value = JSON.stringify(data, null, 2);
          textarea.style.position = 'fixed';
          textarea.style.opacity = '0';
          document.body.appendChild(textarea);
          textarea.select();
          document.execCommand('copy');
          document.body.removeChild(textarea);
        }
      },
    },
    {
      id: 'export-csv',
      label: 'Export',
      icon: Download,
      tooltip: 'Export to CSV',
      onClick: ({ columns, data }) => {
        const headers = columns.join(',');
        const rows = data
          .map((row) =>
            columns
              .map((col) => {
                const value = row[col];
                const str = String(value ?? '');
                if (str.includes(',') || str.includes('"')) {
                  return `"${str.replace(/"/g, '""')}"`;
                }
                return str;
              })
              .join(',')
          )
          .join('\n');
        const csv = `${headers}\n${rows}`;
        const blob = new Blob([csv], { type: 'text/csv' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'query_results.csv';
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
      },
    },
  ];

  // Add "Open in Teradata" action
  const baseUrl = options.teradataBaseUrl || 'http://localhost:8080';
  actions.push({
    id: 'open-in-teradata',
    label: options.projectId ? 'Open in Teradata' : 'New Project',
    icon: ExternalLink,
    tooltip: options.projectId
      ? 'Open this project in Teradata Studio'
      : 'Open in Teradata Studio as a new project',
    onClick: () => {
      const url = options.projectId
        ? `${baseUrl}/project/${options.projectId}`
        : `${baseUrl}/project/new`;
      window.open(url, '_blank');
    },
  });

  return actions;
}

/**
 * Build URL for "Open in Teradata" with SQL context
 */
export function buildTeradataUrl(
  sql: string | undefined,
  projectId: string | undefined,
  teradataBaseUrl: string | undefined
): string {
  const baseUrl = teradataBaseUrl || 'http://localhost:8080';
  const basePath = projectId
    ? `${baseUrl}/project/${projectId}`
    : `${baseUrl}/project/new`;

  if (sql) {
    const encodedSql = encodeURIComponent(sql.slice(0, 1500));
    return `${basePath}?sql=${encodedSql}`;
  }
  return basePath;
}
```

```typescript
/**
 * SQLResultsDisplay - Rich SQL results display with table/chart views
 *
 * REFACTORED: Composes TableWithFilters and ChartWithControls molecules.
 * Reduced from 1772 lines to ~350 lines following atomic design principles.
 *
 * Features:
 * - Table view with sorting, filtering, search via TableWithFilters molecule
 * - Chart view with controls via ChartWithControls molecule
 * - View mode toggle between table and chart
 * - Configurable actions via props
 * - MCP Bridge integration for "Open in Teradata"
 */

import * as React from 'react';
import { AlertCircle } from 'lucide-react';
import { Table as TableIcon, BarChart3 } from 'lucide-react';

import { Badge } from '../../primitives/badge';
import { Button } from '../../primitives/button';
import { ViewModeToggle } from '../../primitives/view-mode-toggle';
import { TableWithFilters } from '../../molecules/table-with-filters';
import { ChartWithControls } from '../../molecules/chart-with-controls';
import { cn } from '../../lib/utils';
import { useMcpBridge } from '../../bridge/McpBridgeProvider';

import type {
  SQLResultsDisplayProps,
  SQLResultsConfig,
  SQLResultsAction,
} from './types';
import { DEFAULT_RESULTS_CONFIG } from './types';
import { createDefaultActions, buildTeradataUrl } from './utils';

export function SQLResultsDisplay({
  data,
  sql,
  savedConfig,
  onConfigChange,
  actions: customActions,
  context = 'web',
  projectId,
  teradataBaseUrl,
  className,
}: SQLResultsDisplayProps) {
  // Merge saved config with defaults
  const initialConfig: SQLResultsConfig = {
    ...DEFAULT_RESULTS_CONFIG,
    ...savedConfig,
    chartConfig: {
      ...DEFAULT_RESULTS_CONFIG.chartConfig,
      ...savedConfig?.chartConfig,
    },
  };

  // State
  const [resultsViewMode, setResultsViewMode] = React.useState<'table' | 'chart'>(
    initialConfig.resultsViewMode
  );
  const [chartHeight, setChartHeight] = React.useState(initialConfig.chartHeight);
  const [tableHeight, setTableHeight] = React.useState(initialConfig.tableHeight);

  const isFirstRender = React.useRef(true);

  // Normalize data - SQL kernel uses 'rows', other formats use 'data'
  const resultData = React.useMemo(
    () => (data.rows || data.data || []) as Record<string, unknown>[],
    [data.rows, data.data]
  );
  const resultColumns = data.columns || [];
  const isSuccess = data.success !== false;

  // Save configuration on changes
  React.useEffect(() => {
    if (isFirstRender.current) {
      isFirstRender.current = false;
      return;
    }

    if (onConfigChange) {
      const config: SQLResultsConfig = {
        ...DEFAULT_RESULTS_CONFIG,
        resultsViewMode,
        chartHeight,
        tableHeight,
      };
      onConfigChange(config);
    }
  }, [resultsViewMode, chartHeight, tableHeight, onConfigChange]);

  // Get bridge for MCP Apps context
  let bridge: ReturnType<typeof useMcpBridge>['bridge'] | null = null;
  try {
    const bridgeContext = useMcpBridge();
    bridge = bridgeContext.bridge;
  } catch {
    // Not inside McpBridgeProvider - that's fine for web context
  }

  // Build "Open in Teradata" URL
  const teradataUrl = React.useMemo(
    () => buildTeradataUrl(sql, projectId, teradataBaseUrl),
    [sql, projectId, teradataBaseUrl]
  );

  // Actions with MCP Bridge support
  const actions = React.useMemo(() => {
    if (customActions) return customActions;

    const baseActions = createDefaultActions({ projectId, teradataBaseUrl });

    // Override open-in-teradata to include SQL and use bridge
    return baseActions.map((action) => {
      if (action.id === 'open-in-teradata') {
        return {
          ...action,
          onClick: async () => {
            if (context === 'mcp-apps' && bridge?.openLink) {
              const result = await bridge.openLink(teradataUrl);
              if (!result.success) {
                try {
                  await navigator.clipboard.writeText(teradataUrl);
                  alert(`Link copied to clipboard: ${teradataUrl}`);
                } catch {
                  alert(`Open this URL: ${teradataUrl}`);
                }
              }
            } else {
              window.open(teradataUrl, '_blank');
            }
          },
        };
      }
      return action;
    });
  }, [customActions, projectId, teradataBaseUrl, context, bridge, teradataUrl]);

  // Error display
  if (!isSuccess && data.error) {
    return (
      <div
        className={cn(
          'bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 rounded-lg p-3',
          className
        )}
        role="alert"
      >
        <div className="flex items-start gap-3">
          <AlertCircle className="w-5 h-5 text-red-600 dark:text-red-400 mt-0.5" />
          <div className="text-sm text-red-700 dark:text-red-400">
            <strong>SQL Error:</strong> {data.error}
          </div>
        </div>
      </div>
    );
  }

  // No results
  if (isSuccess && resultData.length === 0) {
    return (
      <div
        className={cn(
          'bg-green-50 dark:bg-green-900/20 border border-green-200 dark:border-green-800 rounded-lg p-3',
          className
        )}
      >
        <div className="text-sm text-green-700 dark:text-green-400">
          Query executed successfully but returned no results.
        </div>
      </div>
    );
  }

  return (
    <div className={cn('space-y-2', className)}>
      {/* Controls Header */}
      <div className="flex items-center justify-between gap-2 flex-wrap">
        <div className="flex items-center gap-2">
          <Badge variant="outline" className="text-xs">
            {data.row_count || resultData.length || 0} rows
          </Badge>
        </div>
        <div className="flex items-center gap-2 flex-wrap">
          {/* Action Buttons */}
          {actions.map((action) => {
            const Icon = action.icon;
            return (
              <Button
                key={action.id}
                size="sm"
                variant={action.variant || 'outline'}
                onClick={() =>
                  action.onClick({ columns: resultColumns, data: resultData })
                }
                disabled={action.disabled}
                className="h-7 px-2 text-xs"
                title={action.tooltip}
              >
                {Icon && <Icon className="w-3 h-3" />}
                {action.id === 'open-in-teradata' && (
                  <span className="ml-1">{action.label}</span>
                )}
              </Button>
            );
          })}

          {/* View Mode Toggle */}
          <ViewModeToggle
            options={[
              { id: 'table', label: 'Table', icon: TableIcon, title: 'Table view' },
              { id: 'chart', label: 'Chart', icon: BarChart3, title: 'Chart view' },
            ]}
            activeMode={resultsViewMode}
            onModeChange={(newMode) =>
              setResultsViewMode(newMode as 'table' | 'chart')
            }
          />
        </div>
      </div>

      {/* Table View - Using TableWithFilters molecule */}
      {resultsViewMode === 'table' && resultColumns.length > 0 && resultData.length > 0 && (
        <TableWithFilters
          data={resultData}
          columns={resultColumns}
          height={tableHeight}
          showFilter={true}
          showDensityToggle={true}
          virtualScrolling={resultData.length > 100}
        />
      )}

      {/* Chart View - Using ChartWithControls molecule */}
      {resultsViewMode === 'chart' && resultColumns.length > 0 && resultData.length > 0 && (
        <ChartWithControls
          data={resultData}
          columns={resultColumns}
          height={chartHeight}
          showControls={true}
        />
      )}
    </div>
  );
}

// Re-export utility for backward compatibility
export { createDefaultActions } from './utils';
```

```typescript
/**
 * json-render Component Registry
 *
 * Maps catalog definitions to React component implementations.
 * Updated with real component implementations from primitives.

import { defineRegistry } from '@json-render/react';
import { catalog } from './schema';

// Import real component implementations
import { Chart } from '../primitives/chart';
import { DataTable } from '../primitives/data-table';
import { Filter } from '../primitives/filter';
import { DensityToggle } from '../primitives/density-toggle';
import { ViewModeToggle } from '../primitives/view-mode-toggle';

// Wrapper components to adapt props between catalog schema and component props
// These ensure the catalog interface matches what each component expects

function CatalogChart(props: {
  type: string;
  xAxis: string;
  yAxis: string;
  theme?: string | null;
  height?: number;
  showDataLabels?: boolean;
  data?: Record<string, unknown>[];
}) {
  const { data = [], type, xAxis, yAxis, theme, height, showDataLabels } = props;
  return (
    <Chart
      data={data}
      type={type as any}
      xAxis={xAxis}
      yAxis={yAxis}
      theme={theme || undefined}
      height={height}
      showDataLabels={showDataLabels}
    />
  );
}

function CatalogDataTable(props: {
  columns: string[];
  height?: number;
  density?: 'compact' | 'comfortable' | 'spacious';
  virtualScrolling?: boolean;
  data?: Record<string, unknown>[];
}) {
  const { data = [], columns, height, density, virtualScrolling } = props;
  return (
    <DataTable
      data={data}
      columns={columns}
      height={height}
      density={density}
      virtualScrolling={virtualScrolling}
    />
  );
}

function CatalogFilter(props: {
  placeholder?: string;
  columns: string[];
}) {
  // Filter needs value and onChange which come from parent context
  // This is a simplified version for catalog rendering
  const [value, setValue] = React.useState('');
  return (
    <Filter
      value={value}
      onChange={setValue}
      placeholder={props.placeholder}
    />
  );
}

function CatalogDensityToggle(props: {
  value?: 'compact' | 'comfortable' | 'spacious';
}) {
  const [density, setDensity] = React.useState(props.value || 'comfortable');
  return (
    <DensityToggle
      value={density}
      onChange={setDensity}
    />
  );
}

function CatalogViewModeToggle(props: {
  modes?: ('table' | 'chart')[];
  activeMode?: 'table' | 'chart';
}) {
  const { modes = ['table', 'chart'], activeMode: initialMode = 'table' } = props;
  const [activeMode, setActiveMode] = React.useState(initialMode);

  const options = modes.map((mode) => ({
    id: mode,
    label: mode.charAt(0).toUpperCase() + mode.slice(1),
    icon: mode === 'table' ? TableIcon : BarChartIcon,
  }));

  return (
    <ViewModeToggle
      options={options}
      activeMode={activeMode}
      onModeChange={(mode) => setActiveMode(mode as 'table' | 'chart')}
    />
  );
}

// Need React and icons for wrapper components
import * as React from 'react';
import { Table as TableIcon, BarChart3 as BarChartIcon } from 'lucide-react';

export const registry = defineRegistry(catalog, {
  Chart: CatalogChart,
  DataTable: CatalogDataTable,
  Filter: CatalogFilter,
  DensityToggle: CatalogDensityToggle,
  ViewModeToggle: CatalogViewModeToggle,
});
```
